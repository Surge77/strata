# The dashboard's files, compiled into strata_dashboard so that a binary serves
# them with nothing on disk. Each file is embedded as a raw string literal at
# configure time; editing one re-runs CMake (CMAKE_CONFIGURE_DEPENDS).

# file, served path, content type -- three entries per file.
set(_strata_dashboard_files
    index.html    "/"              "text/html; charset=utf-8"
    dashboard.css "/dashboard.css" "text/css; charset=utf-8"
    dashboard.js  "/dashboard.js"  "text/javascript; charset=utf-8"
    model.js      "/model.js"      "text/javascript; charset=utf-8"
    charts.js     "/charts.js"     "text/javascript; charset=utf-8")
set(_delimiter "strata_dashboard_file")

set(_generated "${CMAKE_CURRENT_BINARY_DIR}/dashboard_files.cpp")
set(_staging "${_generated}.staging")
file(WRITE "${_staging}"
     "// Generated by tools/dashboard/CMakeLists.txt. Do not edit.\n"
     "#include \"dashboard_files.hpp\"\n\n"
     "namespace strata::dashboard {\n\n"
     "std::vector<StaticFile> Files() {\n"
     "  return {\n")
list(LENGTH _strata_dashboard_files _count)
math(EXPR _last "${_count} - 1")
foreach(_i RANGE 0 ${_last} 3)
  math(EXPR _path_index "${_i} + 1")
  math(EXPR _type_index "${_i} + 2")
  list(GET _strata_dashboard_files ${_i} _file)
  list(GET _strata_dashboard_files ${_path_index} _path)
  list(GET _strata_dashboard_files ${_type_index} _type)
  set(_source "${CMAKE_CURRENT_SOURCE_DIR}/${_file}")
  file(READ "${_source}" _content)
  string(FIND "${_content}" ")${_delimiter}\"" _clash)
  if(NOT _clash EQUAL -1)
    message(FATAL_ERROR "${_file} contains the raw string delimiter ${_delimiter}")
  endif()
  file(APPEND "${_staging}"
       "      {.path = \"${_path}\",\n"
       "       .content_type = \"${_type}\",\n"
       "       .body = R\"${_delimiter}(${_content})${_delimiter}\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source}")
endforeach()
file(APPEND "${_staging}" "  };\n}\n\n}  // namespace strata::dashboard\n")
# Only a changed file is rewritten, so reconfiguring does not force a rebuild.
configure_file("${_staging}" "${_generated}" COPYONLY)

add_library(strata_dashboard STATIC "${_generated}")
target_include_directories(strata_dashboard PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(strata_dashboard PUBLIC strata)
